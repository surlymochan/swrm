usage: openswarm batch [-h] [--repo REPO] [--cards CARDS] [--hosts HOSTS]
                       [--per-host PER_HOST]
                       [--backend {codex,ocgo,deepseek,cursor,agy,pi,hermes,copilot}]
                       [--model MODEL] [--effort EFFORT] [--verify VERIFY]
                       [--budget-minutes BUDGET_MINUTES]
                       [--backends-file BACKENDS_FILE] [--base BASE]
                       [--scope {repo,host}] [--detach]
                       [--card-type CARD_TYPE] [--rework-of REWORK_OF]
                       [--paths PATHS [PATHS ...]]
                       [{status,plan}] [id]

positional arguments:
  {status,plan}
  id

options:
  -h, --help            show this help message and exit
  --repo REPO
  --cards CARDS
  --hosts HOSTS
  --per-host PER_HOST
  --backend {codex,ocgo,deepseek,cursor,agy,pi,hermes,copilot}
  --model MODEL
  --effort EFFORT
  --verify VERIFY
  --budget-minutes BUDGET_MINUTES
  --backends-file BACKENDS_FILE
                        custom backends config (JSON, M3 slice 1, #101);
                        defaults to $OPENSWARM_HOME/backends.json; passed
                        through to each run
  --base BASE
  --scope {repo,host}   task scope (repo by default; host for authorized host
                        operations)
  --detach
  --card-type CARD_TYPE
                        optional task card type from knowledge/card-types.yaml
  --rework-of REWORK_OF
                        run id this batch reworks
  --paths PATHS [PATHS ...]
                        path hints for matching applicable lessons
